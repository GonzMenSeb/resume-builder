\documentclass[11pt,letterpaper]{article}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}

((* if config.font_family == 'serif' *))

%% Color scheme (max 3 colors per research guidelines)

    linkcolor=primary,
    urlcolor=primary,
    pdfborder={0 0 0}
}

\pagestyle{empty}

%% Section formatting: 14pt headers with rule
\titleformat{\section}
    {\fontsize{14}{16}\bfseries\color{primary}}
    {}
    {0pt}
    {\MakeUppercase}
    [\vspace{-6pt}\textcolor{primary}{\titlerule[0.8pt]}]

\titlespacing*{\section}{0pt}{12pt}{6pt}[0pt]

%% Prevent widow lines after section headers
\makeatletter
\patchcmd{\@afterheading}{\clubpenalty \@M}{\clubpenalty \@M \@afterindentfalse \interlinepenalty\@M}{}{}
\makeatother

%% Bullet list: compact, professional
\setlist[itemize]{
    leftmargin=1.2em,
    itemsep=1pt,
    parsep=0pt,
    topsep=3pt,
    label=\textcolor{primary}{\textbullet}
}
